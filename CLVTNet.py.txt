class ConvStem(nn.Module):
    def __init__(self, in_chans, out_chans, filter_size=3):
        super().__init__()
        self.conv1 = nn.Conv2d(in_chans, out_chans, kernel_size=filter_size, stride=2, padding=1)
        self.conv2 = nn.Conv2d(out_chans, out_chans, kernel_size=filter_size, stride=2, padding=1)
        self.conv3 = nn.Conv2d(out_chans, out_chans, kernel_size=filter_size, stride=1, padding=1)
        self.conv4 = nn.Conv2d(out_chans, out_chans, kernel_size=filter_size, stride=1, padding=1)
        self.conv5 = nn.Conv2d(out_chans, out_chans, kernel_size=1, stride=1)
        self.act = nn.ReLU()
    
    def forward(self, x):
        x = self.act(self.conv1(x))
        x = self.act(self.conv2(x))
        x = self.act(self.conv3(x))
        x = self.act(self.conv4(x))
        x = self.act(self.conv5(x))
        return x
# Patch Embedding Class
class PatchEmbed(nn.Module):
    """
    EEG input data to Patch Embedding
    """

    def __init__(self, img_size=62, patch_size=16, in_c=1, embed_dim=768, norm_layer=None):
        super().__init__()

        self.img_size = img_size
        self.patch_size = patch_size
        self.grid_size = img_size // patch_size  # How many patches in each dimension
        
        self.num_patches = self.grid_size ** 2  # Total patches (assuming square image)
        
        # Convolution to extract patch embeddings
        self.proj = nn.Conv2d(in_c, embed_dim, kernel_size=patch_size, stride=patch_size)
        
        # Normalization layer
        self.norm = norm_layer(embed_dim) if norm_layer else nn.Identity()

    def forward(self, x):
        # x shape: (B, C, H, W) -> B: batch_size, C: channels, H: height, W: width
        x = self.proj(x)  # (B, embed_dim, grid_size, grid_size)
        x = x.flatten(2)  # Flatten spatial dimensions (B, embed_dim, num_patches)
        x = x.transpose(1, 2)  # (B, num_patches, embed_dim)
        x = self.norm(x)  # Apply normalization if needed
        return x
class MLP(nn.Module):
    def __init__(self, input_dim, hidden_dim, output_dim):
        super().__init__()
        self.fc1 = nn.Linear(input_dim, hidden_dim)
        self.fc2 = nn.Linear(hidden_dim, hidden_dim)
        self.fc3 = nn.Linear(hidden_dim, output_dim)
        self.act = nn.ReLU()
    
    def forward(self, x):
        x = self.act(self.fc1(x))
        x = self.act(self.fc2(x))
        x = self.fc3(x)
        return x

# Basic Attention Mechanism (Self-Attention)
class Attention(nn.Module):
    def __init__(self, embed_dim, num_heads):
        super().__init__()
        self.multihead_attn = nn.MultiheadAttention(embed_dim, num_heads)
        self.norm = nn.LayerNorm(embed_dim)
    
    def forward(self, x):
        # x should have shape (seq_len, batch, embed_dim)
        x_attn, _ = self.multihead_attn(x, x, x)
        return self.norm(x + x_attn)


class LiteConvBlock(nn.Module):
    def __init__(self, dim, filter_size=3):
        super().__init__()
        self.dwconv = nn.Conv2d(dim, dim, kernel_size=filter_size, stride=1, padding=1, groups=dim)
        self.pwconv = nn.Conv2d(dim, dim, kernel_size=1, stride=1)
        self.act = nn.ReLU()
    
    def forward(self, x):
        x = self.act(self.dwconv(x))
        x = self.act(self.pwconv(x))
        return x

class CPE(nn.Module):
    def __init__(self, dim, filter_size=3):
        super().__init__()
        self.dwconv = nn.Conv2d(dim, dim, filter_size, padding=1, groups=dim)
    
    def forward(self, x):
        return x + self.dwconv(x)

class GlobalBranch(nn.Module):
    def __init__(self, dim, num_heads, reduction=2, filter_size=3):
        super().__init__()
        self.reduction = reduction
        self.mhsa = nn.MultiheadAttention(dim, num_heads)
        self.ctr = nn.Conv2d(dim, dim, kernel_size=filter_size, stride=reduction, groups=dim)
    
    def forward(self, x):
        B, C, H, W = x.shape
        x_reduced = self.ctr(x).flatten(2).permute(2, 0, 1)  # (N, B, C)
        x = x.flatten(2).permute(2, 0, 1)  # (N, B, C)
        x, _ = self.mhsa(x, x_reduced, x_reduced)
        return x.permute(1, 2, 0).reshape(B, C, H, W)

class LocalBranch(nn.Module):
    def __init__(self, dim, filter_size=3):
        super().__init__()
        self.dwconv = nn.Conv2d(dim, dim, filter_size, padding=1, groups=dim)
        self.spatial_attn = nn.Conv2d(2, 1, kernel_size=7, padding=3)
    
    def forward(self, x):
        pooled = torch.cat([x.mean(dim=1, keepdim=True), x.max(dim=1, keepdim=True)[0]], dim=1)
        attn = torch.sigmoid(self.spatial_attn(pooled))
        return self.dwconv(x) * attn

class FusionModule(nn.Module):
    def __init__(self, dim, reduction=4):
        super().__init__()
        self.fc1 = nn.Linear(dim * 2, dim // reduction)
        self.fc2 = nn.Linear(dim // reduction, dim * 2)
        self.act = nn.ReLU()
        self.sigmoid = nn.Sigmoid()
    
    def forward(self, x_local, x_global):
        x = torch.cat([x_local, x_global], dim=1)  # Concatenate along channel dimension
        x = self.act(self.fc1(x.mean(dim=[2, 3], keepdim=True)))
        x = self.sigmoid(self.fc2(x))
        return x_local * x[:, :x_local.shape[1]] + x_global * x[:, x_local.shape[1]:]

class SPENBlock(nn.Module):
    def __init__(self, dim, num_heads, filter_size=3):
        super().__init__()
        self.cpe = ConvStem(dim, dim, filter_size=filter_size)
        self.global_branch = GlobalBranch(dim, num_heads, filter_size=filter_size)
        self.local_branch = LiteConvBlock(dim, filter_size=filter_size)
        self.fusion = FusionModule(dim)
        self.norm = nn.LayerNorm(dim)
    
    def forward(self, x):
        x = self.cpe(x)
        x_local = self.local_branch(x)
        x_global = self.global_branch(x)
        x = self.fusion(x_local, x_global)
        return self.norm(x)

class SPEN (nn.Module):
    def __init__(self, img_size=256, in_chans=3, num_classes=2, embed_dim=[32, 80, 160, 256], num_heads=[2, 5, 10, 16], filter_sizes=[3, 5, 7, 9], blocks=[2, 2, 5, 2]):
        super().__init__()
        self.conv_stem = ConvStem(in_chans, embed_dim[0], filter_size=filter_sizes[0])
        self.layers = nn.ModuleList([SPENBlock(dim, num_heads[i], filter_size=filter_sizes[i]) for i, dim in enumerate(embed_dim)])
        self.head = nn.Linear(embed_dim[-1], num_classes)
    
    def forward(self, x):
        x = self.conv_stem(x)
        for layer in self.layers:
            x = layer(x)
        x = x.mean(dim=[2, 3])  # Global average pooling
        return self.head(x)

# Assuming EEG data in image form (e.g., spectrograms stored as images)
# Each image is a 2D representation of the EEG signal, with dimensions (channels, height, width)
class EEGDataset(Dataset):
    def __init__(self, data_path, transform=None):
        self.data_path = data_path
        self.transform = transform
        self.X = np.load(os.path.join(data_path, 'X.npy'))  # EEG images (shape: num_samples, num_channels, height, width)
        self.y = np.load(os.path.join(data_path, 'y.npy'))  # Labels (shape: num_samples,)
    
    def __len__(self):
        return len(self.X)
    
    def __getitem__(self, idx):
        sample = self.X[idx]
        label = self.y[idx]
        
        if self.transform:
            sample = self.transform(sample)
        
        return torch.tensor(sample, dtype=torch.float32), torch.tensor(label, dtype=torch.long)

# Training function
def train(model, train_loader, criterion, optimizer, device):
    model.train()
    running_loss = 0.0
    correct = 0
    total = 0
    for inputs, labels in train_loader:
        inputs, labels = inputs.to(device), labels.to(device)

        optimizer.zero_grad()
        
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        
        optimizer.step()
        
        running_loss += loss.item()
        
        _, predicted = torch.max(outputs, 1)
        total += labels.size(0)
        correct += (predicted == labels).sum().item()
    
    avg_loss = running_loss / len(train_loader)
    accuracy = correct / total
    return avg_loss, accuracy

# Validation function
def validate(model, val_loader, criterion, device):
    model.eval()
    running_loss = 0.0
    correct = 0
    total = 0
    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to(device), labels.to(device)

            outputs = model(inputs)
            loss = criterion(outputs, labels)
            running_loss += loss.item()
            
            _, predicted = torch.max(outputs, 1)
            total += labels.size(0)
            correct += (predicted == labels).sum().item()
    
    avg_loss = running_loss / len(val_loader)
    accuracy = correct / total
    return avg_loss, accuracy

# Main function
def main():
    # Parameters
    data_path = 'path_to_your_data'  # Update with the correct path
    batch_size = 32
    num_epochs = 20
    learning_rate = 0.001
    num_classes = 2  # Adjust based on your dataset
    img_size = 256
    embed_dim = [32, 80, 160, 256]
    num_heads = [2, 5, 10, 16]
    filter_sizes = [3, 5, 7, 9]
    blocks = [2, 2, 5, 2]
    
    # Load dataset
    dataset = EEGDataset(data_path)
    train_data, val_data = train_test_split(dataset, test_size=0.2, random_state=42)
    
    train_loader = DataLoader(train_data, batch_size=batch_size, shuffle=True)
    val_loader = DataLoader(val_data, batch_size=batch_size, shuffle=False)
    
    # Model, loss, optimizer
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = SPEN(num_classes=num_classes, embed_dim=embed_dim, num_heads=num_heads, filter_sizes=filter_sizes, blocks=blocks).to(device)
    
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(model.parameters(), lr=learning_rate)
    
    # Training loop
    best_val_accuracy = 0.0
    for epoch in range(num_epochs):
        print(f"Epoch {epoch+1}/{num_epochs}")
        
        # Train the model
        train_loss, train_accuracy = train(model, train_loader, criterion, optimizer, device)
        print(f"Train Loss: {train_loss:.4f}, Train Accuracy: {train_accuracy:.4f}")
        
        # Validate the model
        val_loss, val_accuracy = validate(model, val_loader, criterion, device)
        print(f"Validation Loss: {val_loss:.4f}, Validation Accuracy: {val_accuracy:.4f}")
        
        # Save model if validation accuracy improves
        if val_accuracy > best_val_accuracy:
            best_val_accuracy = val_accuracy
            torch.save(model.state_dict(), 'best_model.pth')
            print("Model saved!")

if __name__ == '__main__':
    main()
